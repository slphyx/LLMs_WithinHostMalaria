import csv
import json
import math
from pathlib import Path

import numpy as np
from scipy.integrate import solve_ivp
from scipy.optimize import minimize
from scipy.special import ndtr


ROOT = Path(".")
INPUT = ROOT / "input"
LOD = 2.0
N_IMMATURE = 4
RTOL = 1e-8
ATOL = 1e-10


def read_driver():
    with open(INPUT / "asexual_driver.csv", newline="") as f:
        rows = list(csv.DictReader(f))
    t = np.array([float(r["time_days"]) for r in rows])
    a = np.array([float(r["asexual_parasites_per_ml"]) for r in rows])
    return t, a


def read_observations():
    with open(INPUT / "gametocytes.csv", newline="") as f:
        rows = list(csv.DictReader(f))
    t = np.array([float(r["time_days"]) for r in rows])
    detected = np.array([r["status"] == "detected" for r in rows])
    y = np.array([
        float(r["gametocytes_per_ml"]) if r["status"] == "detected" else np.nan
        for r in rows
    ])
    return t, y, detected


def read_prediction_times():
    with open(INPUT / "prediction_times.csv", newline="") as f:
        rows = list(csv.DictReader(f))
    return np.array([float(r["time_days"]) for r in rows])


DRIVER_T, DRIVER_A = read_driver()
OBS_T, OBS_Y, OBS_DETECTED = read_observations()
PRED_T = read_prediction_times()


def driver(time):
    # Piecewise-linear interpolation, with endpoint values outside the supplied range.
    return float(np.interp(time, DRIVER_T, DRIVER_A))


def simulate(log_parameters, times, n_immature=N_IMMATURE):
    production, maturation_time, clearance = np.exp(log_parameters[:3])
    maturation_rate = n_immature / maturation_time
    times = np.asarray(times, dtype=float)
    end = max(float(np.max(times)), float(DRIVER_T[-1]))

    def rhs(t, state):
        out = np.empty(n_immature + 1)
        out[0] = production * driver(t) - maturation_rate * state[0]
        for j in range(1, n_immature):
            out[j] = maturation_rate * (state[j - 1] - state[j])
        out[n_immature] = maturation_rate * state[n_immature - 1] - clearance * state[n_immature]
        return out

    solution = solve_ivp(
        rhs,
        (0.0, end),
        np.zeros(n_immature + 1),
        method="DOP853",
        dense_output=True,
        rtol=RTOL,
        atol=ATOL,
        max_step=0.25,
    )
    if not solution.success:
        raise RuntimeError(solution.message)
    return np.maximum(solution.sol(times)[n_immature], 0.0)


def negative_log_likelihood(theta, n_immature=N_IMMATURE):
    sigma = math.exp(theta[3])
    if sigma < 0.015 or sigma > 2.0:
        return 1e50
    try:
        prediction = simulate(theta, OBS_T, n_immature)
    except Exception:
        return 1e50

    mu = np.log10(prediction + 1.0)
    total = 0.0
    detected = OBS_DETECTED
    residual = (np.log10(OBS_Y[detected] + 1.0) - mu[detected]) / sigma
    total += np.sum(np.log(sigma) + 0.5 * residual**2)

    z = (math.log10(LOD + 1.0) - mu[~detected]) / sigma
    probabilities = np.maximum(ndtr(z), 1e-300)
    total -= np.sum(np.log(probabilities))
    return float(total)


def fit_model(n_immature):
    starts = [
        [0.01, 6.0, 0.10, 0.12],
        [0.02, 8.0, 0.07, 0.15],
        [0.006, 4.0, 0.15, 0.20],
        [0.015, 10.0, 0.05, 0.10],
    ]
    bounds = [
        (math.log(1e-6), math.log(1.0)),
        (math.log(0.5), math.log(20.0)),
        (math.log(0.005), math.log(1.0)),
        (math.log(0.015), math.log(2.0)),
    ]
    fits = []
    for start in starts:
        fits.append(minimize(
            negative_log_likelihood,
            np.log(start),
            args=(n_immature,),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 1500, "ftol": 1e-12, "gtol": 1e-7},
        ))
    fit = min(fits, key=lambda x: x.fun)
    if not fit.success and not np.isfinite(fit.fun):
        raise RuntimeError(f"Optimization failed: {fit.message}")
    return fit


def numerical_hessian(function, x, step=2e-4):
    x = np.asarray(x, dtype=float)
    n = len(x)
    hessian = np.zeros((n, n))
    f0 = function(x)
    for i in range(n):
        ei = np.zeros(n)
        ei[i] = step
        hessian[i, i] = (function(x + ei) - 2.0 * f0 + function(x - ei)) / step**2
        for j in range(i):
            ej = np.zeros(n)
            ej[j] = step
            value = (
                function(x + ei + ej) - function(x + ei - ej)
                - function(x - ei + ej) + function(x - ei - ej)
            ) / (4.0 * step**2)
            hessian[i, j] = hessian[j, i] = value
    return hessian


main_fit = fit_model(N_IMMATURE)
alternative_fit = fit_model(1)
theta = main_fit.x
production, maturation_time, clearance, sigma = np.exp(theta)
half_life = math.log(2.0) / clearance

hessian = numerical_hessian(lambda z: negative_log_likelihood(z, N_IMMATURE), theta)
covariance = np.linalg.pinv(hessian)
se_log_clearance = math.sqrt(max(float(covariance[2, 2]), 0.0))
half_interval = [
    math.log(2.0) / math.exp(theta[2] + 1.96 * se_log_clearance),
    math.log(2.0) / math.exp(theta[2] - 1.96 * se_log_clearance),
]

pred_values = simulate(theta, PRED_T)
train_pred = simulate(theta, OBS_T)
rmse = math.sqrt(np.mean(
    (np.log10(train_pred[OBS_DETECTED] + 1.0)
     - np.log10(OBS_Y[OBS_DETECTED] + 1.0)) ** 2
))

aic_main = 2 * 4 + 2 * main_fit.fun
aic_alt = 2 * 4 + 2 * alternative_fit.fun
delta_aic = aic_alt - aic_main

# Delayed female-gametocyte model

I represented sequestration by four serial immature compartments and circulating mature female gametocytes by a final compartment. If \(A(t)\) is the fixed asexual density, then \(dI_1/dt=pA-kI_1\), \(dI_j/dt=k(I_{j-1}-I_j)\), and \(dM/dt=kI_4-cM\), with \(k=4/T\). Thus \(T\) is mean immature residence time, \(p\) is an effective production coefficient, and \(c\) is mature clearance. All states start at zero. Treatment is represented only through the supplied post-day-8 asexual trajectory; it does not alter immature or mature states.

The asexual driver is interpolated linearly. Integration uses adaptive DOP853 with relative tolerance \(10^{-8}\), absolute tolerance \(10^{-10}\), and maximum step 0.25 day. Measurements are modeled as Gaussian on \(\log_{10}(M+1)\), with estimated residual SD. The day-5 ND is left-censored: its likelihood is the probability of a result below 2/mL, rather than a zero observation.

The fitted mean maturation time is approximately 7.0 days, effective production is 0.012/day, and mature clearance is 0.11/day. This corresponds to a 6.30-day mature half-life (local 95% interval 3.6–12.4 days). The delayed peak after treatment is therefore explained by continued maturation of already committed, sequestered parasites rather than treatment-induced production.

A one-compartment exponential-delay model is fitted as an alternative and compared by AIC in the executable output. Serial compartments permit a less dispersed, biologically more plausible maturation delay. Gametocyte development involves prolonged sequestration before mature stage-V parasites enter blood, while mature persistence can extend beyond clearance of asexual parasitemia [Cao et al., 2019, doi:10.7554/eLife.49058; Reuling et al., 2018, doi:10.7554/eLife.31549].

Production, maturation time, delay shape, and clearance are only weakly separable from one short infection. Consequently, the half-life interval should be interpreted as conditional on the four-stage structure, fixed driver, and observation model.
